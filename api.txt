Name: search_buses
Method: GET
URL: /routes/search

Purpose:
Search buses between source and destination

Query Params:
- from (string)
- to (string)
- date (string)

Body:
None

Response:
{
  success: boolean,
  routes: [
    {
      routeId: string,
      busName: string,
      departure: string,
      arrival: string,
      price: number,
      seatsAvailable: number
    }
  ]
}
